-- Lista dinâmica encadeada{
    >> Cada elemento é alocado no momento da inserção, liberado no momento da remoção 
    e guarda o endereço do próximo elemento.

    >> Cada elemento é uma struct com dois campos: informação e ponteiro para a próxima posição
    >> O último elemento aponta para NULL

    struct elemento {
        struct aluno dados; #campo de informação
        struct elemento *prox;  #ponteiro para o próximo nó
    };
    typedef struct elemento Elemento;

    -- VANTAGENS{
        >> Melhor utilização dos recursos de memória.
        >> O tamanho máximo não precisa ser definido previamente.
        >> Inserção e remoção não movimentam os demais elementos.

    }

    -- LIMITAÇÕES{
        >> Acesso indireto aos elementos, sempre por ponteiros.
        >> Necessidade de percorrer a lista para acessar determinado elemento.

    }


}
